create_ing(ing0, "eggs")
create_ing(ing1, "water")
create_tool(t0, "pot")
create_tool(t1, "stove")
create_tool(t2, "bowl with cold water")
create_tool(t3, "knife")
create_tool(t4, "wire or egg slicer")

combine({ing0, ing1}, ing2, "eggs in water", "add enough water to cover the eggs")
put(ing2, t0)
set(t0, "covered with lid")
put(t0, t1)
cook(ing2, t1, ing3, "boiling eggs", "bring to a boil over high heat")
set(t1, "low heat")
cook(ing3, t1, ing4, "hard‑boiled eggs", "simmer for 10-12 minutes")
remove(ing4, t0)
put(ing4, t2)
leave(ing4, "cool in water")
do(ing4, t2, ing5, "peeled eggs", "peel eggs in water")
cut(ing5, t3, ing6, "sliced eggs", "slice the eggs")
cut(ing5, t4, ing7, "sliced eggs", "slice the eggs with wire or egg slicer")
