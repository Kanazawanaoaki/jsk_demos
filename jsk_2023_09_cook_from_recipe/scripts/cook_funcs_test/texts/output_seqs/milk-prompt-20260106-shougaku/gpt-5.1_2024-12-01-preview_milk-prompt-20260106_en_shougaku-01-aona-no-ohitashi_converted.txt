create_ing(ing0, "leafy greens with soil at base")
create_ing(ing1, "stored or running water")
create_ing(ing2, "cold water")

create_tool(t0, "sink or washing bowl")
create_tool(t1, "large pot")
create_tool(t2, "bowl for cold water")
create_tool(t3, "knife")

do(ing0, t0, ing3, "washed greens", "rinse the leaf creases and soil from the base using stored or running water")
set(t1, "fill with generous amount of water")
cook(ing1, t1, ing4, "boiling water", "boil vigorously over high heat")
chefcheck(ing4, "water boiling vigorously")
combine({ing3, ing4}, ing5, "greens in boiling water", "add the greens starting from the base")
chefcheck(ing5, "water boils again")
do(ing5, t1, ing6, "blanched greens", "turn off heat once it boils again")
put(ing2, t2)
do(ing6, t2, ing7, "shocked greens", "plunge the greens into cold water")
do(ing7, , ing8, "squeezed greens", "gently squeeze out excess water while keeping stems aligned")
cut(ing8, t3, ing9, "cut greens (4-5 cm lengths)", "cut into 4-5 cm lengths")
