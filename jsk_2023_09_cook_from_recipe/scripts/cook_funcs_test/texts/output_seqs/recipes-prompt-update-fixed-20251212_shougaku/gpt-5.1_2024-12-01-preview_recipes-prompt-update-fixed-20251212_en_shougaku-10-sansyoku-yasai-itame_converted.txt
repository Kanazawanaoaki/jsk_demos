wash(carrot)
wash(bell-pepper)
wash(cabbage)

cut(bell-pepper, julienned-5mm, knife)
cut(cabbage, julienned-1cm-x-4cm, knife)
pour(oil, frying-pan)
heat(oil, medium-hot)
stir-fry(oil, coated-pan, frying-pan)
pour(carrot, frying-pan)
stir-fry(carrot, quickly-stir-fried, spatula)
pour(bell-pepper, frying-pan)
stir-fry(bell-pepper, quickly-stir-fried, spatula)
pour(cabbage, frying-pan)
stir-fry(cabbage, quickly-stir-fried, spatula)
do(vegetables, seasoned-with-salt-and-pepper, frying-pan, None)
turn-off-stove(frying-pan)
do(vegetables, plated, plate, None)
