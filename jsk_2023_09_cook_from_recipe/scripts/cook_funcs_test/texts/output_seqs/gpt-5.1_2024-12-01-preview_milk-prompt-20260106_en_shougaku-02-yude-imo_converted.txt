create_ing(ing0, "potatoes")
create_ing(ing1, "water")
create_tool(t0, "sink")
create_tool(t1, "knife")
create_tool(t2, "pot")
create_tool(t3, "stove")
create_tool(t4, "colander")
create_tool(t5, "bamboo skewer")

do(ing0, t0, ing2, "washed potatoes", "wash off any dirt thoroughly")
chefcheck(ing2, "no visible dirt; avoid any potatoes that have turned green")
do(ing2, t1, ing3, "sprout-removed potatoes", "remove the sprouts")
cut(ing3, t1, ing4, "uniform potato pieces", "cut into uniform sizes")
combine({ing4, ing1}, ing5, "potatoes in water", "add enough water to cover the potatoes")
put(ing5, t2)
put(ing2, t2)
set(t3, "high heat")
cook(ing5, t3, ing6, "boiling potatoes", "bring to a boil")
set(t3, "low heat")
cook(ing6, t3, ing7, "simmered potatoes", "cook until tender")
chefcheck(ing7, "bamboo skewer pierces easily")
put(ing7, t4)
remove(ing7, t2)
leave(ing7, "cool until safe to handle")
do(ing7, t1, ing8, "peeled potatoes", "peel after they have cooled; be careful not to burn yourself")
